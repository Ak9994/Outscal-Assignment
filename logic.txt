C# code for level based spawn:
[Serializable]
public class Wave {
    public Enemy enemyPrefab;
    public int enemyCount;
    public float spawnInterval;
}

public class EnemySpawner : MonoBehaviour {
    public List<Wave> waves;
    private int currentWave = 0;

    IEnumerator Start() {
        while (currentWave < waves.Count) {
            Wave wave = waves[currentWave];
            for (int i = 0; i < wave.enemyCount; i++) {
                Instantiate(wave.enemyPrefab, transform.position, Quaternion.identity);
                yield return new WaitForSeconds(wave.spawnInterval);
            }
            // wait until all enemies are gone
            while (GameObject.FindGameObjectsWithTag("Enemy").Length > 0)
                yield return null;
            currentWave++;
        }
        Debug.Log("All waves complete!");
    }
} 



interface AmmoManager{
    int CurrentAmmo();     //This is used to get current ammo
    int ReserveAmmo();     //This is used to get final leftover ammo 
    void Ammo_used(int n); //This represents ammo used
    void Ammo_Left(int n); //This represents ammo left
    bool Reload();         // Boolean Check to see if reload is needed or not
    void CalculateAmmo(int n); //Calculate the ammo needed to reload
}

class Magazines: AmmoManager{
    private:
        int Current;
        int Reserve;
        int Capacity;
    public:
        Magazines(int cap,int res){
            Current = cap;
            Capacity = cap;
            Reserve = res;
        }
        void Ammo_Left(int no) => Current = no;
        void Ammo_used(int no) => Current = Math.Max(1,Current-no);
        void CalculateAmmo(int no){
            int need_ammo = Math.Max(1,no-Current);
            int left_ammo = Math.Min(need_ammo,Reserve);
            Current += left_ammo;
            Reserve -= left_ammo;
        }
};
